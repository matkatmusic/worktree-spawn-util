```typescript
export async function isBranchMerged(
  repoRoot: string,
  branch: string,
  parentBranch: string,
): Promise<boolean> {
  const git = simpleGit(repoRoot);
  const merged = await git.raw(["branch", "--merged", parentBranch]);
  const mergedBranches = merged.split("\n").map(l => l.replace(/^\*?\s+/, "").trim());
  return mergedBranches.includes(branch);
}
```

```typescript
export async function hasNewCommits(

  branch: string,
  parentCommit: string,
): Promise<boolean> {
  const git = simpleGit(repoRoot);
  const log = await git.raw(["log", `${parentCommit}..${branch}`, "--oneline"]);
  return log.trim().length > 0;
}
```

```typescript
export async function onlyIgnorableChanges(worktreePath: string): Promise<boolean> {
  const git = simpleGit(worktreePath);
  const status = await git.raw(["status", "--porcelain"]);
  const lines = status.trim().split("\n").filter((l) => l.trim().length > 0);
  if (lines.length === 0) return true;
  return lines.every((line) => {
    const filePath = line.slice(3).split(" -> ")[0];
    return filePath.startsWith(".claude/") || filePath.startsWith(".vscode/");
  });
}
```

**Key Finding**

Scenario 5 does **not** work reliably as implemented.

The Git logic is mostly right for a normal merge, but your `isBranchMerged()` parser is wrong for linked worktrees. `git branch` marks branches checked out in linked worktrees with `+`, and your regex only strips `*`, so `+ testv1` will not match `"testv1"`. Git documents both behaviors: `--merged <commit>` lists branches whose tip is reachable from that commit, and linked-worktree branches are shown with a plus sign.  
Sources: [git-branch](https://git-scm.com/docs/git-branch)

**Answers**

1. `git branch --merged master`:
For a normal merge or fast-forward merge, yes, Git will consider `testv1` merged because its tip is reachable from `master`. But while the worktree still exists, the output is likely `+ testv1`, so **your implementation can return false even when Git says it is merged**.  
Sources: [git-branch](https://git-scm.com/docs/git-branch)

2. `git log abc123..testv1 --oneline` after merge:
Yes. `abc123..testv1` means “commits reachable from `testv1` excluding those reachable from `abc123`.” Merging into `master` does not move `testv1`, so those commits still appear.  
Sources: [gitrevisions](https://git-scm.com/docs/gitrevisions)

3. Edge cases:
- Fast-forward merge: works by reachability; `testv1` is merged.
- True merge commit: works by reachability; `testv1` is merged.
- Squash merge: fails your merge test by design; Git says squash does not move `HEAD` to the topic tip or record merge ancestry, so `testv1` is **not** merged by reachability.
- Rebase/cherry-pick/rebase-and-merge style integration: same problem as squash. Equivalent changes may be on `master`, but the original `testv1` tip is not reachable, so `--merged master` is false.
- Worktree state after merging on `master`: the linked worktree is not auto-updated. Git worktrees share refs, but `HEAD`, index, and working tree state are per-worktree. So merging in the main repo does not rewrite files in `/.../.worktrees/testv1`.  
Sources: [git-merge](https://git-scm.com/docs/git-merge), [git-worktree](https://git-scm.com/docs/git-worktree)

4. `git worktree remove --force testv1`:
Yes, removing a linked worktree whose branch is checked out there is normal. Git only cares that the worktree is clean unless `--force` is used, and locked worktrees need `--force` twice. One correction: `<worktree>` is a **path or unique path suffix**, not a branch name.  
Sources: [git-worktree](https://git-scm.com/docs/git-worktree)

5. `git branch -d testv1` after removal:
Yes, **after** the worktree is removed, as long as Git considers the branch fully merged. Before removal, branch deletion will fail because the branch is checked out in another worktree. Also, `-d` requires merged into upstream or `HEAD` if no upstream is set.  
Sources: [git-branch](https://git-scm.com/docs/git-branch)

6. `simpleGit(repoRoot)` before vs after worktree removal:
Read-only ref queries like `log`, `merge-base`, and `branch --merged` work fine either way because refs are shared across worktrees. The important difference is that while the linked worktree exists, Git tracks `testv1` as checked out elsewhere, which affects branch listing and branch mutation/deletion safeguards. After removal, that occupancy metadata is gone.  
Sources: [git-worktree](https://git-scm.com/docs/git-worktree)

**Recommendation**

Do not parse `git branch --merged` human output. Use:

```bash
git merge-base --is-ancestor testv1 master
```

Git documents `--is-ancestor` directly, and it avoids the `+ testv1` parsing bug.  
Source: [git-merge-base](https://git-scm.com/docs/git-merge-base)
